/**
 * 示例：把 store 类库适配成 yoya 状态引擎（zustand/vanilla）。
 * 复制到自己项目，把 import 换成你要用的库即可。
 * store 形态：一个值单元就是一个极小 store（getState / setState / subscribe）；
 * batch 是必需方法，需要自己收拢通知（见末尾）。
 */
import { createStore } from 'zustand/vanilla';

export function createZustandAdapter() {
  const listeners = new WeakMap(); // source -> Set<listener>
  const pending = new Map(); // source -> value（批量期间合并）
  let depth = 0;

  const notify = (source, value) => {
    const set = listeners.get(source);
    if (set) {
      set.forEach((listener) => listener(value));
    }
  };

  return {
    name: 'zustand-example',

    createSignal: (initial) => createStore(() => ({ value: initial })),

    read: (source) => source.getState().value,

    write(source, value) {
      if (Object.is(source.getState().value, value)) {
        return;
      }

      source.setState({ value });
      if (depth > 0) {
        pending.set(source, value);
      } else {
        notify(source, value);
      }
    },

    subscribe(source, listener) {
      let set = listeners.get(source);
      if (!set) {
        set = new Set();
        listeners.set(source, set);
      }

      set.add(listener);
      return () => set.delete(listener);
    },

    batch(run) {
      depth += 1;
      try {
        return run();
      } finally {
        depth -= 1;
        if (depth === 0) {
          const jobs = [...pending];
          pending.clear();
          jobs.forEach(([source, value]) => notify(source, value));
        }
      }
    }
  };
}
